import React from 'react';
import { Filter, RefreshCcw, Search } from 'lucide-react';
import type {
  AdjustmentDomain,
  AdjustmentStatus,
  AdjustmentTargetType,
} from '../../services/adjustment.service';
import {
  adjustmentDomainMeta,
  adjustmentStatusMeta,
} from './adjustment.constants';

interface AdjustmentFiltersProps {
  query: string;
  onQueryChange: (value: string) => void;
  domainFilter: 'all' | AdjustmentDomain;
  onDomainFilterChange: (value: 'all' | AdjustmentDomain) => void;
  statusFilter: 'all' | AdjustmentStatus;
  onStatusFilterChange: (value: 'all' | AdjustmentStatus) => void;
  targetTypeFilter: 'all' | AdjustmentTargetType;
  onTargetTypeFilterChange: (value: 'all' | AdjustmentTargetType) => void;
  onRefresh: () => void;
}

const AdjustmentFilters = ({
  query,
  onQueryChange,
  domainFilter,
  onDomainFilterChange,
  statusFilter,
  onStatusFilterChange,
  targetTypeFilter,
  onTargetTypeFilterChange,
  onRefresh,
}: AdjustmentFiltersProps) => {
  return (
    <div className="bg-white/80 dark:bg-slate-900/80 backdrop-blur-xl rounded-[36px] border border-slate-100 dark:border-slate-800 p-5 lg:p-6 shadow-sm">
      <div className="flex flex-col gap-4 xl:flex-row xl:items-center xl:justify-between">
        <div className="flex flex-wrap items-center gap-2">
          <div className="inline-flex items-center gap-2 px-4 py-2 rounded-full bg-slate-100 dark:bg-slate-800 text-[10px] font-black uppercase tracking-widest">
            <Filter size={12} />
            过滤
          </div>
          {(['all', 'finance', 'production', 'inventory'] as const).map(item => (
            <button
              key={item}
              onClick={() => onDomainFilterChange(item)}
              className={`px-4 py-2 rounded-full text-[10px] font-black uppercase tracking-widest border ${
                domainFilter === item
                  ? 'bg-slate-900 text-white border-slate-900 dark:bg-white dark:text-slate-900 dark:border-white'
                  : 'bg-transparent text-slate-500 border-slate-200 dark:border-slate-700 hover:bg-slate-50 dark:hover:bg-slate-800'
              }`}
            >
              {item === 'all' ? '全部域' : adjustmentDomainMeta[item].label}
            </button>
          ))}
          {(['all', 'pending', 'posted', 'reversed', 'rejected'] as const).map(item => (
            <button
              key={item}
              onClick={() => onStatusFilterChange(item)}
              className={`px-4 py-2 rounded-full text-[10px] font-black uppercase tracking-widest border ${
                statusFilter === item
                  ? 'bg-blue-600 text-white border-blue-600'
                  : 'bg-transparent text-slate-500 border-slate-200 dark:border-slate-700 hover:bg-slate-50 dark:hover:bg-slate-800'
              }`}
            >
              {item === 'all' ? '全部状态' : adjustmentStatusMeta[item].label}
            </button>
          ))}
          {(['all', 'order', 'productBatch', 'manual'] as const).map(item => (
            <button
              key={item}
              onClick={() => onTargetTypeFilterChange(item)}
              className={`px-4 py-2 rounded-full text-[10px] font-black uppercase tracking-widest border ${
                targetTypeFilter === item
                  ? 'bg-emerald-600 text-white border-emerald-600'
                  : 'bg-transparent text-slate-500 border-slate-200 dark:border-slate-700 hover:bg-slate-50 dark:hover:bg-slate-800'
              }`}
            >
              {item === 'all' ? '全部对象' : item === 'order' ? '订单' : item === 'productBatch' ? '批次' : '手工'}
            </button>
          ))}
        </div>
        <div className="flex items-center gap-3">
          <div className="flex items-center gap-2 px-4 py-3 rounded-[20px] bg-slate-50 dark:bg-slate-800 border border-slate-100 dark:border-slate-700 min-w-[260px]">
            <Search size={16} className="text-slate-400" />
            <input
              value={query}
              onChange={e => onQueryChange(e.target.value)}
              placeholder="搜索单号、原因、对象..."
              className="w-full bg-transparent outline-none text-sm font-bold text-slate-700 dark:text-slate-100 placeholder:text-slate-400"
            />
          </div>
          <button
            onClick={onRefresh}
            className="inline-flex items-center gap-2 px-5 py-3 rounded-[20px] bg-slate-900 text-white text-[10px] font-black uppercase tracking-widest shadow-lg shadow-slate-900/10"
          >
            <RefreshCcw size={14} />
            刷新
          </button>
        </div>
      </div>
    </div>
  );
};

export default AdjustmentFilters;
