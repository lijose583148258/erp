import React from 'react';
import { Layers3, PencilLine, Sparkles } from 'lucide-react';
import type { AdjustmentFormState, AdjustmentTemplate } from './adjustment.constants';
import type {
  AdjustmentDomain,
  AdjustmentStatus,
  AdjustmentTargetType,
} from '../../services/adjustment.service';

interface AdjustmentFormProps {
  form: AdjustmentFormState;
  setForm: React.Dispatch<React.SetStateAction<AdjustmentFormState>>;
  templates: readonly AdjustmentTemplate[];
  onCreate: () => void;
  onReset: () => void;
}

const AdjustmentForm = ({ form, setForm, templates, onCreate, onReset }: AdjustmentFormProps) => {
  return (
    <div className="bg-white dark:bg-slate-900 rounded-[36px] border border-slate-100 dark:border-slate-800 shadow-sm p-6 lg:p-7">
      <div className="flex items-start justify-between gap-4">
        <div>
          <h3 className="text-xl font-black italic uppercase tracking-tighter text-slate-900 dark:text-white">
            新建调账
          </h3>
          <p className="text-[10px] font-black uppercase tracking-widest text-slate-400 mt-2">
            财务补录、生产损耗、库存盘差
          </p>
        </div>
        <div className="p-3 rounded-2xl bg-slate-100 dark:bg-slate-800 text-slate-500">
          <PencilLine size={18} />
        </div>
      </div>

      <div className="mt-5 grid grid-cols-3 gap-2">
        {templates.map(tpl => {
          const Icon = tpl.icon;
          return (
            <button
              key={tpl.id}
              onClick={() => setForm(prev => ({ ...prev, ...tpl.patch }))}
              className="rounded-[18px] border border-slate-100 dark:border-slate-800 bg-slate-50 dark:bg-slate-800/60 p-3 text-left hover:border-blue-200 dark:hover:border-blue-800"
            >
              <div className="flex items-center gap-2 text-[10px] font-black uppercase tracking-widest text-slate-500">
                <Icon size={12} />
                {tpl.label}
              </div>
              <div className="mt-2 text-[10px] text-slate-400 leading-5">{tpl.helperText}</div>
            </button>
          );
        })}
      </div>

      <div className="mt-4 grid grid-cols-2 gap-2">
        <select
          value={form.domain}
          onChange={e =>
            setForm(prev => ({
              ...prev,
              domain: e.target.value as AdjustmentDomain,
              targetType: e.target.value === 'finance' ? 'order' : 'productBatch',
            }))
          }
          className="px-4 py-3 rounded-2xl bg-slate-50 dark:bg-slate-800 text-xs font-black border border-slate-100 dark:border-slate-700"
        >
          <option value="finance">财务</option>
          <option value="production">生产</option>
          <option value="inventory">库存</option>
        </select>
        <select
          value={form.status}
          onChange={e => setForm(prev => ({ ...prev, status: e.target.value as AdjustmentStatus }))}
          className="px-4 py-3 rounded-2xl bg-slate-50 dark:bg-slate-800 text-xs font-black border border-slate-100 dark:border-slate-700"
        >
          <option value="posted">直接生效</option>
          <option value="pending">先登记</option>
        </select>
      </div>

      <div className="mt-2 grid grid-cols-2 gap-2">
        <select
          value={form.targetType}
          onChange={e => setForm(prev => ({ ...prev, targetType: e.target.value as AdjustmentTargetType }))}
          className="px-4 py-3 rounded-2xl bg-slate-50 dark:bg-slate-800 text-xs font-black border border-slate-100 dark:border-slate-700"
        >
          <option value="order">订单</option>
          <option value="productBatch">批次</option>
          <option value="manual">手工</option>
        </select>
        <input
          value={form.targetRef}
          onChange={e => setForm(prev => ({ ...prev, targetRef: e.target.value }))}
          placeholder="对象参考号"
          className="px-4 py-3 rounded-2xl bg-slate-50 dark:bg-slate-800 text-xs font-bold border border-slate-100 dark:border-slate-700"
        />
      </div>

      <div className="mt-2 grid grid-cols-2 gap-2">
        <input
          value={form.orderId}
          onChange={e => setForm(prev => ({ ...prev, orderId: e.target.value }))}
          placeholder="订单ID"
          className={`px-4 py-3 rounded-2xl bg-slate-50 dark:bg-slate-800 text-xs font-bold border border-slate-100 dark:border-slate-700 ${
            form.domain !== 'finance' ? 'opacity-60' : ''
          }`}
        />
        <input
          value={form.batchId}
          onChange={e => setForm(prev => ({ ...prev, batchId: e.target.value }))}
          placeholder="批次ID"
          className={`px-4 py-3 rounded-2xl bg-slate-50 dark:bg-slate-800 text-xs font-bold border border-slate-100 dark:border-slate-700 ${
            form.domain === 'finance' ? 'opacity-60' : ''
          }`}
        />
      </div>

      <div className="mt-2 grid grid-cols-2 gap-2">
        <input
          value={form.amountDelta}
          onChange={e => setForm(prev => ({ ...prev, amountDelta: e.target.value }))}
          placeholder="金额变动"
          className={`px-4 py-3 rounded-2xl bg-slate-50 dark:bg-slate-800 text-xs font-bold border border-slate-100 dark:border-slate-700 ${
            form.domain !== 'finance' ? 'opacity-60' : ''
          }`}
        />
        <input
          value={form.quantityDelta}
          onChange={e => setForm(prev => ({ ...prev, quantityDelta: e.target.value }))}
          placeholder="数量变动"
          className={`px-4 py-3 rounded-2xl bg-slate-50 dark:bg-slate-800 text-xs font-bold border border-slate-100 dark:border-slate-700 ${
            form.domain === 'finance' ? 'opacity-60' : ''
          }`}
        />
      </div>

      <div className="mt-2 grid grid-cols-2 gap-2">
        <input
          value={form.reasonCategory}
          onChange={e => setForm(prev => ({ ...prev, reasonCategory: e.target.value }))}
          placeholder="原因分类"
          className="px-4 py-3 rounded-2xl bg-slate-50 dark:bg-slate-800 text-xs font-bold border border-slate-100 dark:border-slate-700"
        />
        <input
          value={form.lossType}
          onChange={e => setForm(prev => ({ ...prev, lossType: e.target.value }))}
          placeholder="损耗类型"
          className="px-4 py-3 rounded-2xl bg-slate-50 dark:bg-slate-800 text-xs font-bold border border-slate-100 dark:border-slate-700"
        />
      </div>

      <textarea
        value={form.reason}
        onChange={e => setForm(prev => ({ ...prev, reason: e.target.value }))}
        placeholder="调账原因"
        rows={3}
        className="mt-2 w-full px-4 py-3 rounded-2xl bg-slate-50 dark:bg-slate-800 text-xs font-bold border border-slate-100 dark:border-slate-700 resize-none"
      />
      <textarea
        value={form.note}
        onChange={e => setForm(prev => ({ ...prev, note: e.target.value }))}
        placeholder="备注"
        rows={2}
        className="mt-2 w-full px-4 py-3 rounded-2xl bg-slate-50 dark:bg-slate-800 text-xs font-bold border border-slate-100 dark:border-slate-700 resize-none"
      />

      <div className="mt-3 grid grid-cols-2 gap-3">
        <button
          onClick={onCreate}
          className="inline-flex items-center justify-center gap-2 px-4 py-3 rounded-2xl bg-blue-600 text-white text-[10px] font-black uppercase tracking-[0.25em]"
        >
          <Sparkles size={14} />
          创建单据
        </button>
        <button
          onClick={onReset}
          className="inline-flex items-center justify-center gap-2 px-4 py-3 rounded-2xl bg-slate-100 dark:bg-slate-800 text-slate-600 dark:text-slate-200 text-[10px] font-black uppercase tracking-[0.25em]"
        >
          <Layers3 size={14} />
          重置
        </button>
      </div>
    </div>
  );
};

export default AdjustmentForm;
